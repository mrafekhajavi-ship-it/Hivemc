socket.on('positions_update', (positions) => { // تبدیل کلیدها به حروف کوچک برای اینکه اشتباه تایپی پیش نیاید const lowerPositions = {}; for (let k in positions) { lowerPositions[k.toLowerCase()] = positions[k]; } const myPos = lowerPositions[username.toLowerCase()]; if (myPos) { statusDiv.innerHTML = `موقعیت شما: X:${myPos.x} Y:${myPos.y} Z:${myPos.z} (جهان: ${myPos.world})`; } else { statusDiv.innerHTML = `در انتظار دریافت موقعیت شما از بازی...`; } // بررسی صدا برای هر بازیکن for (const [targetName, audio] of Object.entries(peersAudio)) { const targetPos = lowerPositions[targetName.toLowerCase()]; if (!myPos || !targetPos) { // اگر هنوز مختصات یکی از بازیکنان از بازی نیامده، صدا را موقت کامل پخش کن تا تست شود! audio.volume = 1.0; continue; } if (myPos.world !== targetPos.world) { audio.volume = 0; continue; } // محاسبه فاصله ۳ بعدی const dx = myPos.x - targetPos.x; const dy = myPos.y - targetPos.y; const dz = myPos.z - targetPos.z; const distance = Math.sqrt(dx * dx + dy * dy + dz * dz); const MAX_DIST = 25.0; // حداکثر فاصله مجاز if (distance >= MAX_DIST) { audio.volume = 0; } else { // هرچه نزدیک‌تر، صدا بلندتر const vol = Math.max(0, Math.min(1, 1 - (distance / MAX_DIST))); audio.volume = vol; } } });
